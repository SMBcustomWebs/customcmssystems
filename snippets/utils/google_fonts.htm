// Get the string of fonts we gathered above $raw_fonts = ''; $fonts_array = explode('|', $raw_fonts); // Fonts we DO NOT want to ask Google for $skip_fonts = ['System Default', '-', 'Arial', 'Helvetica', 'Georgia', 'Times New Roman', 'Consolas', 'Courier New', 'Brush Script MT', 'Impact', '']; $final_fonts = []; // Filter out duplicates and web-safe fonts foreach ($fonts_array as $font) { $font = trim($font); if (!in_array($font, $skip_fonts) && !in_array($font, $final_fonts)) { $final_fonts[] = $font; } } $url_parts = []; // Format for the Google API foreach ($final_fonts as $font) { $formatted_font = str_replace(' ', '+', $font); $url_parts[] = 'family=' . $formatted_font . ':ital,wght@0,300..800;1,300..800'; } // Combine them together into the final string $font_url = implode('&', $url_parts); // Pass it back to CouchCMS global $CTX; $CTX->set('final_font_url', $font_url, 'global');
